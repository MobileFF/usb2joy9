cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED ENV{PICO_SDK_PATH} AND NOT DEFINED PICO_SDK_PATH)
  set(PICO_SDK_PATH $ENV{HOME}/dev/pico-sdk)
endif()
include(pico_sdk_import.cmake)
project(pintest C CXX ASM)
pico_sdk_init()
add_executable(pintest pintest.c)
target_link_libraries(pintest PRIVATE pico_stdlib)
pico_enable_stdio_usb(pintest 0)
pico_enable_stdio_uart(pintest 0)
pico_add_extra_outputs(pintest)
